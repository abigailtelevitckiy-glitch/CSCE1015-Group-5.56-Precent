Example 1:O(N^2)

This code gives you about half the points, when N<=10^3. You need to realize that you can only change +/- 1 to the furthest block, and therefore you should change the furthest block first, and you can ignore it after it reaches 0 bacteria level. The code below brute-force the problem


	#include
	using namespace std;
	long long g[200010];
	int main()
	{
		int n;
		cin>>n;
		for(int i=1;i<=n;i++)
		{
			cin>>g[i];
		}
		long long sum=0;
		for(int i=1;i<=n;i++)
		{
			if(g[i]==0)
				continue;
			sum+=abs(g[i]);
			long long k=0;
			long long c=0;
			k=-g[i];
			c=k;
			for(int j=i;j<=n;j++)
			{
				g[j]+=c;
				c+=k;
			}
		}
		cout<<sum;
		return 0;
	}

Example 2:O(N)

This code gives you full points, when N<=2*10^5. After understanding the brute-force code, you need to improve the algorithm to O(N). To do that, you need to store the amount you need to change each block for each operation without looping through everything, instead changing everything with one loop. Below is the code that does this.


	#include
	using namespace std;
	long long g[200010];
	int main()
	{
		int n;
		cin>>n;
		for(int i=1;i<=n;i++)
		{
			cin>>g[i];
		}
		long long sum=0;
		for(int i=1;i<=n;i++)
		{
			if(g[i]==0)
				continue;
			sum+=abs(g[i]);
			long long k=0;
			long long c=0;
			k=-g[i];
			c=k;
			for(int j=i;j<=n;j++)
			{
				g[j]+=c;
				c+=k;
			}
		}
		cout<<sum;
		return 0;
	}
